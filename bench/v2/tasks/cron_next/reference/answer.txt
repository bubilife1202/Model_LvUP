```python
from datetime import datetime, timedelta


FIELD_RANGES = [
    (0, 59),
    (0, 23),
    (1, 31),
    (1, 12),
    (0, 6),
]


def _parse_iso(value):
    return datetime.strptime(value, "%Y-%m-%dT%H:%M:%SZ")


def _format_iso(value):
    return value.strftime("%Y-%m-%dT%H:%M:%SZ")


def _expand_piece(piece, minimum, maximum):
    if "/" in piece:
        base, step_text = piece.split("/", 1)
        step = int(step_text)
        if base == "*":
            start, end = minimum, maximum
        else:
            start_text, end_text = base.split("-", 1)
            start, end = int(start_text), int(end_text)
        return set(range(start, end + 1, step))
    if piece == "*":
        return set(range(minimum, maximum + 1))
    if "-" in piece:
        start_text, end_text = piece.split("-", 1)
        start, end = int(start_text), int(end_text)
        return set(range(start, end + 1))
    return {int(piece)}


def _parse_field(text, minimum, maximum):
    allowed = set()
    for piece in text.split(","):
        allowed.update(_expand_piece(piece, minimum, maximum))
    return sorted(allowed), text != "*"


def _cron_dow(value):
    return (value.weekday() + 1) % 7


def _date_matches(current, dom_allowed, dom_restricted, dow_allowed, dow_restricted):
    dom_match = current.day in dom_allowed
    dow_match = _cron_dow(current) in dow_allowed
    if dom_restricted and dow_restricted:
        return dom_match or dow_match
    return dom_match and dow_match


def next_fire(cron_expr, after_iso):
    minute_field, hour_field, dom_field, month_field, dow_field = cron_expr.split()
    minute_allowed, _ = _parse_field(minute_field, *FIELD_RANGES[0])
    hour_allowed, _ = _parse_field(hour_field, *FIELD_RANGES[1])
    dom_allowed, dom_restricted = _parse_field(dom_field, *FIELD_RANGES[2])
    month_allowed, _ = _parse_field(month_field, *FIELD_RANGES[3])
    dow_allowed, dow_restricted = _parse_field(dow_field, *FIELD_RANGES[4])

    start = _parse_iso(after_iso).replace(second=0, microsecond=0) + timedelta(minutes=1)
    current_date = start.date()

    while True:
        day_anchor = datetime(current_date.year, current_date.month, current_date.day)
        if day_anchor.month in month_allowed and _date_matches(
            day_anchor, dom_allowed, dom_restricted, dow_allowed, dow_restricted
        ):
            for hour in hour_allowed:
                if current_date == start.date() and hour < start.hour:
                    continue
                for minute in minute_allowed:
                    if current_date == start.date() and hour == start.hour and minute < start.minute:
                        continue
                    candidate = datetime(
                        current_date.year,
                        current_date.month,
                        current_date.day,
                        hour,
                        minute,
                    )
                    if candidate >= start:
                        return _format_iso(candidate)
        current_date += timedelta(days=1)
```
